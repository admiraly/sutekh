# A minimal project that includes Sutekh with add_subdirectory, as a game with Sutekh as a git submodule does.
# Driven by tests/consumer/check_consumer.py (and the consumer CI job), which configures it in a fresh build tree and
# checks: ctest lists only this project's tests; a source that includes an engine-internal header fails to compile;
# Sutekh::host links and runs; engine headers reach this project as SYSTEM includes; the Sutekh checkout is not
# written to.
cmake_minimum_required(VERSION 3.21)
project(sutekh_consumer LANGUAGES C)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

get_filename_component(_consumer_default_sutekh "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SUTEKH_SOURCE_DIR "${_consumer_default_sutekh}" CACHE PATH "The Sutekh checkout (a submodule in a real project)")

enable_testing()
# The binary directory argument is needed only because the checkout is outside this source tree; a submodule at
# extern/sutekh is simply add_subdirectory(extern/sutekh).
add_subdirectory("${SUTEKH_SOURCE_DIR}" sutekh)

# The consumer's own strict warning set; engine headers must not trip it (they are SYSTEM includes).
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    set(CONSUMER_WARNINGS -Wall -Wextra -Wpedantic -Werror)
    set(CONSUMER_STRICT_EXTRA -Wpadded) # fires on engine structs when the headers are not SYSTEM
elseif(MSVC)
    set(CONSUMER_WARNINGS /W4 /WX)
    set(CONSUMER_STRICT_EXTRA "")
endif()

# (c) Sutekh::host links and runs: a world from the repository's movement example, 120 ticks, rollback check.
add_executable(consumer_host consumer_host.c)
target_compile_options(consumer_host PRIVATE ${CONSUMER_WARNINGS})
target_link_libraries(consumer_host PRIVATE Sutekh::host)
add_test(NAME consumer_host COMMAND consumer_host "${SUTEKH_SOURCE_DIR}")

# Every public header in one translation unit, with the strict set plus CONSUMER_STRICT_EXTRA, linked against every
# public library alias that exists in this configuration.
file(GLOB _consumer_public_headers RELATIVE "${SUTEKH_SOURCE_DIR}/include" CONFIGURE_DEPENDS
     "${SUTEKH_SOURCE_DIR}/include/sutekh/*.h")
set(_consumer_tu "/* generated by tests/consumer/CMakeLists.txt */\n")
foreach(h IN LISTS _consumer_public_headers)
    string(APPEND _consumer_tu "#include \"${h}\"\n")
endforeach()
string(APPEND _consumer_tu "int main(void) { return su_host_api_version() > 0u ? 0 : 1; }\n")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/consumer_public_headers.c" CONTENT "${_consumer_tu}")
add_executable(consumer_public_headers "${CMAKE_CURRENT_BINARY_DIR}/consumer_public_headers.c")
target_compile_options(consumer_public_headers PRIVATE ${CONSUMER_WARNINGS} ${CONSUMER_STRICT_EXTRA})
set(_consumer_libs "")
foreach(t IN LISTS SUTEKH_PUBLIC_TARGETS)
    get_target_property(_type ${t} TYPE)
    if(NOT _type STREQUAL "EXECUTABLE" AND NOT t STREQUAL "Sutekh::host_shared")
        list(APPEND _consumer_libs ${t})
    endif()
endforeach()
target_link_libraries(consumer_public_headers PRIVATE ${_consumer_libs})
add_test(NAME consumer_public_headers COMMAND consumer_public_headers)

# (b) Engine-internal headers are not reachable through Sutekh::host: each probe must fail to compile. Not in `all`;
# the driver builds them one by one and checks the failure is the missing header.
foreach(probe src src_platform space3d)
    add_executable(consumer_probe_${probe} EXCLUDE_FROM_ALL internal_probe.c)
    string(TOUPPER "${probe}" _probe_macro)
    target_compile_definitions(consumer_probe_${probe} PRIVATE CONSUMER_PROBE_${_probe_macro}=1)
    target_link_libraries(consumer_probe_${probe} PRIVATE Sutekh::host)
endforeach()

# What the driver checks against the build tree.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/consumer_facts.txt"
     "sutekh_public_targets=${SUTEKH_PUBLIC_TARGETS}\n"
     "sutekh_build_tests=${SUTEKH_BUILD_TESTS}\n"
     "sutekh_build_bench=${SUTEKH_BUILD_BENCH}\n"
     "sutekh_build_examples=${SUTEKH_BUILD_EXAMPLES}\n"
     "sutekh_werror=${SUTEKH_WERROR}\n"
     "sutekh_vulkan_include_dir=${SUTEKH_VULKAN_INCLUDE_DIR}\n"
     "c_compiler_id=${CMAKE_C_COMPILER_ID}\n")
